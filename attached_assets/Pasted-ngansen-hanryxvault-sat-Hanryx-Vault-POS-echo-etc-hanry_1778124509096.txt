#!/bin/sh
# HanryxVault kiosk autostart — runs once when labwc starts.
# Must export DISPLAY/XAUTHORITY because chromium uses --ozone-platform=x11
# (XWayland), and setsid+nohup detaches from labwc's env.
(
    sleep 5
    export DISPLAY=:0
    export XAUTHORITY=/home/ngansen/.Xauthority
    export WAYLAND_DISPLAY=wayland-0
    export XDG_RUNTIME_DIR=/run/user/1000
    export XDG_SESSION_TYPE=wayland
    export MOZ_ENABLE_WAYLAND=1
    setsid nohup /bin/bash /home/ngansen/.hanryx-dual-monitor.sh \
        >> /var/log/hanryx-kiosk.log 2>&1 < /dev/null &
) &
