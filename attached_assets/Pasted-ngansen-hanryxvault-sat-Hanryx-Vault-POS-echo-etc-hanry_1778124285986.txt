#!/bin/sh
# HanryxVault kiosk autostart — runs once when labwc starts.
# Must export DISPLAY/XAUTHORITY because chromium uses --ozone-platform=x11
# (XWayland), and setsid+nohup detaches from labwc's env.
(
    sleep 5
    export DISPLAY=:0
    export XAUTHORITY=/home/ngansen/.Xauthority
    export WAYLAND_DISPLAY=wayland-0
    export XDG_RUNTIME_DIR=/run/user/1000
    export XDG_SESSION_TYPE=wayland
    export MOZ_ENABLE_WAYLAND=1
    setsid nohup /bin/bash /home/ngansen/.hanryx-dual-monitor.sh \
        >> /var/log/hanryx-kiosk.log 2>&1 < /dev/null &
) &

#!/bin/sh
# Launched by hanryx-kiosk.service as ngansen. Sets the runtime env labwc
# needs when started outside a logind session (no greeter).
set -e

export HOME="/home/ngansen"
export USER="ngansen"
export LOGNAME="ngansen"
export SHELL="/bin/bash"

# logind/seatd-friendly runtime dir (XDG_RUNTIME_DIR must exist + be 0700)
export XDG_RUNTIME_DIR="/run/user/1000"
if [ ! -d "$XDG_RUNTIME_DIR" ]; then
    mkdir -p "$XDG_RUNTIME_DIR"
    chown "ngansen:ngansen" "$XDG_RUNTIME_DIR"
    chmod 0700 "$XDG_RUNTIME_DIR"
fi

# Kiosk-session env (also re-applied by /etc/hanryx-kiosk/labwc/environment)
export XDG_CURRENT_DESKTOP=labwc:wlroots
export XDG_SESSION_DESKTOP=hanryx-kiosk
export XDG_SESSION_TYPE=wayland
export MOZ_ENABLE_WAYLAND=1

cd "$HOME"
exec /usr/bin/labwc -C /etc/hanryx-kiosk/labwc
